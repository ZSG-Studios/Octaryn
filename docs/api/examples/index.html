Project File

A game module references Octaryn.Shared and only approved packages. Basegame currently uses Arch packages through the allowlist.

<Project Sdk="Microsoft.NET.Sdk">
  <PropertyGroup>
    <GenerateRuntimeConfigurationFiles>true</GenerateRuntimeConfigurationFiles>
  </PropertyGroup>

  <ItemGroup>
    <ProjectReference Include="..\octaryn-shared\Octaryn.Shared.csproj" />
  </ItemGroup>

  <ItemGroup>
    <PackageReference Include="Arch" />
    <PackageReference Include="Arch.System" />
    <PackageReference Include="Arch.System.SourceGenerator"
      PrivateAssets="all"
      IncludeAssets="analyzers;build;buildTransitive"
      OutputItemType="Analyzer" />
    <PackageReference Include="Arch.EventBus" />
    <PackageReference Include="Arch.Relationships" />
  </ItemGroup>

  <ItemGroup>
    <Content Include="Data/Module/*.module.json" CopyToOutputDirectory="PreserveNewest" CopyToPublishDirectory="PreserveNewest" />
    <Content Include="Data/**/*.json" Exclude="Data/Module/*.module.json" CopyToOutputDirectory="PreserveNewest" CopyToPublishDirectory="PreserveNewest" />
    <Content Include="Assets/**/*" Exclude="Assets/**/.gitkeep" CopyToOutputDirectory="PreserveNewest" CopyToPublishDirectory="PreserveNewest" />
  </ItemGroup>
</Project>

Registration And Manifest

The registration object is the module entry point. The manifest must use current property names: RequiredCapabilities, RequestedHostApis, ContentDeclarations, AssetDeclarations, and Schedule.

using Octaryn.Shared.ApiExposure;
using Octaryn.Shared.FrameworkAllowlist;
using Octaryn.Shared.GameModules;
using Octaryn.Shared.Host;
using Octaryn.Shared.World;

namespace Example.Game.Module;

public sealed class ModuleRegistration :
    IGameModuleRegistration,
    IBlockAuthorityRulesProvider,
    IWorldGenerationRulesProvider
{
    public IBlockAuthorityRules BlockAuthorityRules { get; } = new ExampleBlockAuthorityRules();

    public IWorldGenerationRules WorldGenerationRules { get; } = new ExampleWorldGenerationRules();

    public GameModuleManifest Manifest { get; } = new(
        ModuleId: "example.game",
        DisplayName: "Example Game",
        Version: "0.1.0",
        OctarynApiVersion: "0.1.0",
        RequiredCapabilities:
        [
            ModuleCapabilityIds.ContentBlocks,
            ModuleCapabilityIds.GameplayRules,
            ModuleCapabilityIds.GameplayInteractions,
            ModuleCapabilityIds.WorldBlockEdits,
            ModuleCapabilityIds.WorldgenBiomes,
            ModuleCapabilityIds.WorldgenFeatures,
            ModuleCapabilityIds.WorldgenNoise
        ],
        RequestedHostApis:
        [
            HostApiIds.Frame,
            HostApiIds.Commands
        ],
        RequestedRuntimePackages:
        [
            AllowedPackageIds.Arch,
            AllowedPackageIds.ArchSystem,
            AllowedPackageIds.ArchEventBus,
            AllowedPackageIds.ArchRelationships
        ],
        RequestedBuildPackages:
        [
            AllowedPackageIds.ArchSystemSourceGenerator
        ],
        RequestedFrameworkApiGroups:
        [
            FrameworkApiGroupIds.BclPrimitives,
            FrameworkApiGroupIds.BclCollections,
            FrameworkApiGroupIds.BclMemory,
            FrameworkApiGroupIds.BclMath,
            FrameworkApiGroupIds.BclTime,
            FrameworkApiGroupIds.BclText
        ],
        ModuleDependencies: [],
        ContentDeclarations:
        [
            new GameModuleContentDeclaration(
                "example.game.blocks",
                "block",
                "Data/Blocks/example.game.blocks.json"),
            new GameModuleContentDeclaration(
                "example.game.rule.terrain_generation",
                "rule",
                "Data/Rules/example.game.rule.terrain_generation.json")
        ],
        AssetDeclarations:
        [
            new GameModuleAssetDeclaration(
                "example.game.texture.atlas.color",
                "atlas",
                "Assets/Atlases/example-game-color.png")
        ],
        Schedule: new GameModuleScheduleDeclaration(
        [
            ExampleScheduleDeclarations.FrameTick
        ]),
        Compatibility: new GameModuleCompatibility(
            MinimumHostApiVersion: "0.1.0",
            MaximumHostApiVersion: "0.1.0",
            SaveCompatibilityId: "example.game.save.v0",
            SupportsMultiplayer: false));

    public IGameModuleInstance CreateInstance(ModuleHostContext context)
    {
        return ExampleGameContext.Create(context);
    }
}

Runtime Instance

The host calls Tick with frame timing and world time. Store the narrow ModuleHostContext if the module needs command requests.

using Arch.Core;
using Octaryn.Shared.GameModules;

namespace Example.Game;

public sealed class ExampleGameContext : IGameModuleInstance
{
    private readonly ModuleHostContext _host;
    private readonly World _world = World.Create();
    private Entity _frameEntity;
    private bool _hasFrameEntity;

    private ExampleGameContext(ModuleHostContext host)
    {
        _host = host;
    }

    public static ExampleGameContext Create(ModuleHostContext host)
    {
        return new ExampleGameContext(host);
    }

    public void Tick(in ModuleFrameContext frame)
    {
        EnsureFrameEntity();
        _world.Set(_frameEntity, new ExampleFrameState(
            frame.FrameIndex,
            frame.DeltaSeconds,
            frame.WorldTime.TotalSeconds));
    }

    public void Dispose()
    {
        _world.Dispose();
    }

    private void EnsureFrameEntity()
    {
        if (_hasFrameEntity)
        {
            return;
        }

        _frameEntity = _world.Create(new ExampleFrameState(0, 0.0, 0.0));
        _hasFrameEntity = true;
    }
}

public readonly record struct ExampleFrameState(
    ulong FrameIndex,
    double DeltaSeconds,
    double TotalWorldSeconds);

Scheduled Work

Declare host reads and writes up front. Command access must be a scheduled write to host.commands.

using Octaryn.Shared.ApiExposure;
using Octaryn.Shared.Host;

namespace Example.Game.Module;

public static class ExampleScheduleDeclarations
{
    public static ScheduledSystemDeclaration FrameTick { get; } = new(
        SystemId: "example.game.frame_tick",
        Phase: HostWorkPhase.Gameplay,
        FrameOrTickOwner: HostScheduleIds.FrameOrTickOwner,
        Reads:
        [
            new ScheduledResourceAccess(HostApiIds.Frame, ScheduledAccessMode.Read)
        ],
        Writes:
        [
            new ScheduledResourceAccess("example.game.frame_state", ScheduledAccessMode.Write),
            new ScheduledResourceAccess(HostApiIds.Commands, ScheduledAccessMode.Write)
        ],
        RunsAfter: [],
        RunsBefore: [],
        Flags: HostWorkScheduleFlags.DeterministicOrder | HostWorkScheduleFlags.RequiresTickBarrier,
        CommitBarrier: HostScheduleIds.FrameOrTickEndBarrier);
}

Request Block Edits

Modules request edits; server authority decides whether the edit applies and whether it persists. Do not write save files directly from module code.

using Octaryn.Shared.GameModules;
using Octaryn.Shared.World;

public static class ExampleEditRequests
{
    public static bool TryPlaceDebugBlock(ModuleHostContext host, ulong frameIndex)
    {
        var edit = new BlockEdit(
            new BlockPosition(0, 80, 0),
            new BlockId(1));

        return host.Commands.TryRequest(
            ModuleCommandRequest.SetBlock(edit, requestId: frameIndex));
    }

    public static bool TryBreakBlock(ModuleHostContext host, BlockPosition position, ulong frameIndex)
    {
        return host.Commands.TryRequest(
            ModuleCommandRequest.BreakBlock(position, requestId: frameIndex));
    }
}

Block Authority Rules

Authority rules belong to the game module, but the server owns enforcement. Basegame uses these rules to validate known blocks, support, solidity, and client-placeable blocks.

using Octaryn.Shared.World;

namespace Example.Game;

public sealed class ExampleBlockAuthorityRules : IBlockAuthorityRules
{
    public bool IsKnownBlock(BlockId block)
    {
        return block == BlockId.Air || block.Value is >= 1 and <= 128;
    }

    public bool CanApplyEdit(BlockEdit edit, BlockId belowBlock)
    {
        return edit.Block == BlockId.Air || IsSolidBlock(belowBlock);
    }

    public bool CanStaySupported(BlockId block, BlockPosition position, BlockId belowBlock)
    {
        _ = position;
        return block == BlockId.Air || IsSolidBlock(belowBlock);
    }

    public bool IsClientPlaceable(BlockId block)
    {
        return block != BlockId.Air && IsKnownBlock(block);
    }

    public bool IsSolidBlock(BlockId block)
    {
        return block != BlockId.Air && IsKnownBlock(block);
    }
}

World Generation Rules

World generation returns plans and feature edits. The server owns generation and persistence; generated seed terrain is not saved as authoritative edit data.

using Octaryn.Shared.World;

namespace Example.Game;

public sealed class ExampleWorldGenerationRules : IWorldGenerationRules
{
    public int WaterHeight => 30;

    public BlockId WaterBlock => new(16);

    public TerrainMaterialRules Materials => new(
        SandBlock: new BlockId(4),
        GrassBlock: new BlockId(1),
        DirtBlock: new BlockId(2),
        StoneBlock: new BlockId(3),
        SnowBlock: new BlockId(5));

    public TerrainColumnPlan PlanTerrainColumn(TerrainColumnSample sample)
    {
        var height = (int)global::System.Math.Clamp(
            sample.HeightNoise * 50.0f + 30.0f,
            0.0f,
            sample.MaxTerrainY);

        return new TerrainColumnPlan(
            sample.WorldX,
            sample.WorldZ,
            sample.LocalX,
            sample.LocalZ,
            sample.LocalWidth,
            sample.LocalDepth,
            height,
            global::System.Math.Max(height, WaterHeight),
            Materials.GrassBlock,
            Materials.DirtBlock,
            IsLowland: height < 40,
            HasGrassSurface: true);
    }

    public void AddFeatureBlocks(TerrainColumnPlan column, float plantNoise, ICollection<BlockEdit> blocks)
    {
        if (!column.HasGrassSurface || plantNoise < 0.65f)
        {
            return;
        }

        blocks.Add(new BlockEdit(
            new BlockPosition(column.WorldX, column.DecorationY + 1, column.WorldZ),
            new BlockId(10)));
    }
}

Content And Assets

Content declarations must point under Data/. Asset declarations must point under Assets/ or Shaders/. IDs should be owned by the module ID prefix.

example-game/
  Example.Game.csproj
  Source/
    Module/ModuleRegistration.cs
    Module/ExampleScheduleDeclarations.cs
    Gameplay/ExampleGameContext.cs
    Gameplay/ExampleBlockAuthorityRules.cs
    Worldgen/ExampleWorldGenerationRules.cs
  Data/
    Module/example.game.module.json
    Blocks/example.game.blocks.json
    Rules/example.game.rule.terrain_generation.json
  Assets/
    Atlases/example-game-color.png

Validate The Module

Use the owner module validation probe for bundled modules and shared policy checks. A clean compile is not enough; manifests must pass policy validation.

tools/build/cmake_build.sh debug-linux --target octaryn_validate_owner_module_validation_probe
tools/build/cmake_build.sh debug-linux --target octaryn_validate_module_source_api

dotnet run \
  --project tools/validation/Octaryn.ModuleManifestProbe/Octaryn.ModuleManifestProbe.csproj \
  --configuration Debug -- octaryn-basegame

Common validation failures are missing host API declarations, command writes without world.block_edits, content outside Data/, assets outside Assets/ or Shaders/, and denied framework groups.

Recommended Layout

Use the same owner vocabulary as the main modules: Source/, Data/, Assets/, Shaders/, and Tools/ when applicable. Keep gameplay code focused by domain, and keep host internals out of module code.
